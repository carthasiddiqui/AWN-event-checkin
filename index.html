// ── AWN ATTENDANCE TRACKER ── // Watches awneventtracker@gmail.com for check-in CSV emails // and appends attendance data to the master Google Sheet. // Set this to run on a time trigger (every hour). const SHEET_ID = '196nrbNcH-NaXnDHkSOFJilrLYk4ISue2UFQSclPtgSU'; const EMAIL_SUBJECT = 'AWN Check-In Data'; const PROCESSED_LABEL = 'processed'; function importCheckInData() { const sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); // Ensure header row exists if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Event Name', 'Event Date', 'First Name', 'Last Name', 'Email', 'Status', // Attended, No-Show, Late Cancellation, Walk-in 'AWN Member', 'Check-In Time', 'Imported At' ]); sheet.getRange(1, 1, 1, 9).setFontWeight('bold').setBackground('#4a86e8').setFontColor('#ffffff'); sheet.setFrozenRows(1); } // Get or create the 'processed' label let processedLabel = GmailApp.getUserLabelByName(PROCESSED_LABEL); if (!processedLabel) { processedLabel = GmailApp.createLabel(PROCESSED_LABEL); } // Search for unprocessed AWN check-in emails const threads = GmailApp.search(`subject:"${EMAIL_SUBJECT}" -label:${PROCESSED_LABEL}`); if (threads.length === 0) { Logger.log('No new check-in emails found.'); return; } threads.forEach(thread => { const messages = thread.getMessages(); messages.forEach(message => { const attachments = message.getAttachments(); attachments.forEach(attachment => { if (attachment.getName().endsWith('.csv')) { const csvContent = attachment.getDataAsString(); const rows = parseCSV(csvContent); if (rows.length < 2) return; // No data rows const headers = rows[0].map(h => h.trim().toLowerCase()); const eventNameIdx = headers.indexOf('event name'); const eventDateIdx = headers.indexOf('event date'); const firstNameIdx = headers.indexOf('first name'); const lastNameIdx = headers.indexOf('last name'); const emailIdx = headers.indexOf('email'); const statusIdx = headers.indexOf('status'); const awnMemberIdx = headers.indexOf('awn member'); const checkInTimeIdx = headers.indexOf('check-in time'); const importedAt = new Date().toLocaleString(); const dataRows = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (row.every(cell => !cell.trim())) continue; dataRows.push([ eventNameIdx >= 0 ? row[eventNameIdx] : '', eventDateIdx >= 0 ? row[eventDateIdx] : '', firstNameIdx >= 0 ? row[firstNameIdx] : '', lastNameIdx >= 0 ? row[lastNameIdx] : '', emailIdx >= 0 ? row[emailIdx] : '', statusIdx >= 0 ? row[statusIdx] : '', awnMemberIdx >= 0 ? row[awnMemberIdx] : '', checkInTimeIdx >= 0 ? row[checkInTimeIdx] : '', importedAt ]); } if (dataRows.length > 0) { sheet.getRange(sheet.getLastRow() + 1, 1, dataRows.length, 9).setValues(dataRows); Logger.log(`Imported ${dataRows.length} rows from: ${attachment.getName()}`); } } }); }); // Mark thread as processed thread.addLabel(processedLabel); Logger.log(`Marked thread as processed: ${thread.getFirstMessageSubject()}`); }); Logger.log('Import complete.'); } // ── CSV PARSER ── function parseCSV(text) { const lines = text.trim().split(/\r?\n/); return lines.map(line => { const result = []; let cur = '', inQuote = false; for (let i = 0; i < line.length; i++) { const c = line[i]; if (c === '"') { if (inQuote && line[i + 1] === '"') { cur += '"'; i++; } else inQuote = !inQuote; } else if (c === ',' && !inQuote) { result.push(cur.trim()); cur = ''; } else cur += c; } result.push(cur.trim()); return result; }); }
